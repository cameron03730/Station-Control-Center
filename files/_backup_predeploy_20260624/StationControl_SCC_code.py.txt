def _user(session):
    """Authenticated username from the Perspective session."""
    return session.props.auth.user.userName


def _logOverride(stationName, overrideType, overrideDetails, reason, overrideUser):
    """
    Reuse the EXISTING audit query. One row per committed action.
    (Supervisor Overrides/insert/insertEntrySupervisorOverrideLog)
    """
    system.db.runNamedQuery(
        project=_PROJECT,
        path="station-control/Supervisor Overrides/insert/insertEntrySupervisorOverrideLog",
        parameters={"stationName": stationName,
                    "overrideType": overrideType,
                    "overrideDetails": overrideDetails,
                    "reason": reason,
                    "overrideUser": overrideUser})


def _backend(stationName=None):
    """
    Owning backend gateway name: Frame/Boom/Paint (TW/TT/PT) -> BE1, else BE2.
    For shared-SCADA-DB named queries the choice is interchangeable; routed by
    area for consistency with the existing getNextWorkOrder pattern.
    """
    prefix = (stationName or "")[:2].upper()
    if prefix in ("TW", "TT", "PT"):
        return GatewayConfig.getBackEnd_1()
    return GatewayConfig.getBackEnd_2()


def _beExec(functionPath, params, stationName=None):
    """
    Invoke an EXISTING FactoryGatewayEvents script function on the owning backend
    via the established 'executeFunction' message handler. The handler dispatches
    with kwargs:  eval(payload['function'])(**payload['params']).
    Every BE call in this file goes through here so the sendRequest plumbing lives
    in exactly one place.
    """
    return system.util.sendRequest(
        "FactoryGatewayEvents", "executeFunction",
        payload={"function": functionPath, "params": params},
        remoteServer=str(_backend(stationName)))


def _beQuery(queryPath, queryParams, stationName=None):
    """
    Run a FactoryGatewayEvents named query on a backend. The FE does NOT host the
    FactoryGatewayEvents project, so those queries are executed remotely via the
    existing 'executeFunction' message handler -> JcCommonOperations.dynamicQuery.
    Thin wrapper over _beExec.
    BE contract:  JcCommonOperations.dynamicQuery(queryPath, queryParams)  [MUST BE
    ADDED on the BE -- see CLAUDE.md / redundancy notes].
    """
    return _beExec("JcCommonOperations.dynamicQuery",
                   {"queryPath": queryPath, "queryParams": queryParams},
                   stationName=stationName)


def getTagFolder(stationName):
    """Resolve a station's tag folder using the EXISTING canonical resolver."""
    return StationControl.DynamicView.getTagFolder(stationName)

def stationComponentIDTagPath(stationName, leaf="componentID"):
    """
    fromIGN component-ID tag path via the EXISTING DynamicView.getTagFolder resolver
    (returns the area folder; the station name is appended to reach the station).
    'leaf' is 'componentID' (default) or 'componentID_2' for two-component stations.
    """
    folder = getTagFolder(stationName)
    if not folder:
        return None
    return "%s/%s/control/fromIGN/%s" % (folder, stationName, leaf)


def getGatewayForStation(stationName):
    """'BE1' for Frame/Boom/Paint (TAGIO1), else 'BE2'. Display only."""
    prefix = (stationName or "")[:2].upper()
    return "BE1" if prefix in ("TW", "TT", "PT") else "BE2"


def runBoundaryChecks(stationName, newComponentID):
    """
    Read-only structured boundary checks for the Station tab UI (chip + list).
    Uses ONLY the assembly WIP status (no PLC) and the AMR-bound tag read, plus
    the existing MachineConfig lookup. Returns {'checks':[...], 'allPassed':bool}.
    """
    logger.info("runBoundaryChecks called with stationName=%r newComponentID=%r" % (stationName, newComponentID))
    checks = []
    newVal = (newComponentID or "").strip()
    isClear = (newVal == "")

    active = stationActiveWipCount(stationName)
    checks.append({"label": "Not actively building (no WIP row, status 2)",
                   "passed": active == 0,
                   "detail": "OK" if active == 0 else "%d active WIP row(s)" % active})

    bound, detail = stationAmrBound(stationName)
    checks.append({"label": "No AMR currently bound",
                   "passed": not bound,
                   "detail": "OK" if not bound else ("AMR %s bound" % detail)})

    if isClear:
        checks.append({"label": "New ComponentID exists in MachineConfig",
                       "passed": True, "detail": "N/A - clearing"})
    else:
        mc = JcCommonOperations.getMachineConfigInfo(newVal, ["ItemName"])
        ok = (mc is not None and len(mc) > 0)
        checks.append({"label": "New ComponentID exists in MachineConfig",
                       "passed": ok, "detail": "OK" if ok else "not found"})

    current = readStationComponentID(stationName)
    changed = (current is not None) and (str(current).strip() != newVal)
    checks.append({"label": "Value actually changes",
                   "passed": changed, "detail": "OK" if changed else "no change"})

    result = {"checks": checks, "allPassed": all(c["passed"] for c in checks)}
    logger.info("runBoundaryChecks returning %r" % result)
    return result


def readStationComponentID(stationName, leaf="componentID"):
    """One-shot read of a station's componentID (or componentID_2). No polling."""
    path = stationComponentIDTagPath(stationName, leaf)
    if path is None:
        logger.warn("readStationComponentID: no tag path resolved for %r" % stationName)
        return None
    qv = system.tag.readBlocking([path])[0]
    if qv is None or not qv.quality.isGood():
        return None
    return "" if qv.value is None else str(qv.value)


def stationActiveWipCount(stationName):
    """
    Boundary-check source (per build direction): WIP status in the assembly table,
    NOT PLC. The SCC does not read PLC status. Uses the new scalar query
    'station-control-center/getActiveStations' (param StationName).
    """
    logger.info("stationActiveWipCount called with stationName=%r" % stationName)
    res = system.db.runNamedQuery(
        project=_PROJECT,
        path="station-control-center/getActiveStations",
        parameters={"StationName": stationName})
    count = int(res)
    logger.info("stationActiveWipCount returning %d for %r" % (count, stationName))
    return count


def stationAmrBound(stationName):
    """True if an AMR is bound (AMRNumberAssignedToStation > 0). Read-only."""
    logger.info("stationAmrBound called with stationName=%r" % stationName)
    folder = getTagFolder(stationName)
    if not folder:
        logger.warn("stationAmrBound: no folder for %r" % stationName)
        return (False, "Unknown station")
    amrPath = "%s/%s/AMRControl/AMRNumberAssignedToStation" % (folder, stationName)
    qv = system.tag.readBlocking([amrPath])[0]
    logger.info("stationAmrBound: tag=%r quality=%r value=%r" % (amrPath, qv.quality, qv.value))
    if qv is None or not qv.quality.isGood() or qv.value is None:
        return (False, "no AMR")
    return (int(qv.value) > 0, str(qv.value))


def commitStationComponentID(stationName, newComponentID, reason, session, leaf="componentID"):
    """
    Set/clear a station's fromIGN componentID (or componentID_2), with checks + audit.
    'leaf' is 'componentID' (default) or 'componentID_2' for two-component stations.
    Returns {'ok':bool,'message':str}.
    """
    slot = "ComponentID_2" if leaf == "componentID_2" else "ComponentID"
    newVal = (newComponentID or "").strip()
    reason = (reason or "").strip()
    if not reason:
        return {"ok": False, "message": "A reason is required."}

    path = stationComponentIDTagPath(stationName, leaf)
    if path is None:
        return {"ok": False, "message": "Could not resolve a tag path for %s." % stationName}

    before = readStationComponentID(stationName, leaf)
    if before is None:
        return {"ok": False, "message": "Could not read current %s value; aborting." % slot}

    # Boundary checks (assembly WIP status only - no PLC)
    if stationActiveWipCount(stationName) > 0:
        return {"ok": False, "message": "Station has an active WIP order (status 2); edit blocked."}
    bound, detail = stationAmrBound(stationName)
    if bound:
        return {"ok": False, "message": "AMR %s is bound; clear it on the AMR tab first." % detail}
    if newVal != "":
        # MachineConfig existence via the existing shared helper
        mc = JcCommonOperations.getMachineConfigInfo(newVal, ["ItemName"])
        inConfig = (mc is not None and len(mc) > 0)
        if not inConfig:
            return {"ok": False, "message": "'%s' not found in MachineConfig." % newVal}
    if str(before).strip() == newVal:
        return {"ok": False, "message": "%s already matches; no change." % slot}

    # Write
    writeValue = None if newVal == "" else newVal
    qc = system.tag.writeBlocking([path], [writeValue])[0]
    if not qc.isGood():
        return {"ok": False, "message": "Tag write returned bad quality."}

    # Audit (existing query)
    isClear = (newVal == "")
    details = "%s: %s -> %s" % (slot,
                                "(blank)" if before == "" else before,
                                "(blank)" if isClear else newVal)
    _logOverride(stationName,
                 "Station %s Clear" % slot if isClear else "Station %s Set" % slot,
                 details, reason, _user(session))
    return {"ok": True, "message": "Committed: %s %s" % (stationName, details)}

def getAmrList():
    """
    Build the AMR list WITHOUT a new named query: browse the existing
    [TAGIO2].../Fleet/AMR Data/ folder once (on demand) and read each AMR's
    two componentID fields. No polling, no per-row subscriptions -
    a single browse + one batched readBlocking.
    Returns a list[dict]: AMR, ComponentID, ComponentID2.
    NOTE: AMR_xxx are UdtInstance (typeId Fleet/AMR/AMR), so DO NOT filter the
    browse by tagType='Folder' (that returns nothing).
    """
    base = "[TAGIO2]JLG/Jefferson City/Fleet/AMR Data"
    results = system.tag.browse(base).getResults()
    names = []
    for r in results:
        name = str(r["name"])           # e.g. AMR_047
        if name.startswith("AMR_"):
            names.append(name)
    if not names:
        return []
    names.sort()

    # (column, tag leaf relative to the AMR instance) - only the two componentID
    # fields, kept lean so the whole-fleet read stays light (~2 tags x N AMRs).
    fields = [("ComponentID", "machineInfo/componentID"),
              ("ComponentID2", "machineInfo/componentID2")]
    paths = []
    for name in names:
        for _, leaf in fields:
            paths.append("%s/%s/%s" % (base, name, leaf))

    vals = system.tag.readBlocking(paths)
    rows = []
    k = 0
    for name in names:
        d = {"AMR": name.replace("AMR_", "")}
        for col, _ in fields:
            qv = vals[k]
            k += 1
            good = qv is not None and qv.quality.isGood() and qv.value is not None
            d[col] = "" if not good else str(qv.value)
        rows.append(d)
    return rows


def amrComponentIDTagPath(amrId):
    """[TAGIO2].../Fleet/AMR Data/AMR_{id:03d}/machineInfo/componentID (existing pattern)."""
    s = str(amrId).zfill(3)
    return "[TAGIO2]JLG/Jefferson City/Fleet/AMR Data/AMR_%s/machineInfo/componentID" % s


def readAmrComponentID(amrId):
    qv = system.tag.readBlocking([amrComponentIDTagPath(amrId)])[0]
    if qv is None or not qv.quality.isGood():
        return None
    return "" if qv.value is None else str(qv.value)


def commitAmrComponentID(amrId, newComponentID, reason, session):
    """Set/clear an AMR's machineInfo/componentID, with audit. Unassign does NOT cascade to the station."""
    newVal = (newComponentID or "").strip()
    reason = (reason or "").strip()
    if not reason:
        return {"ok": False, "message": "A reason is required."}
    path = amrComponentIDTagPath(amrId)
    before = readAmrComponentID(amrId)
    if before is None:
        return {"ok": False, "message": "Could not read AMR_%s; aborting." % str(amrId).zfill(3)}
    if str(before).strip() == newVal:
        return {"ok": False, "message": "New value matches current value."}

    writeValue = None if newVal == "" else newVal
    qc = system.tag.writeBlocking([path], [writeValue])[0]
    if not qc.isGood():
        return {"ok": False, "message": "Tag write returned bad quality."}

    isClear = (newVal == "")
    details = "AMR_%s: %s -> %s" % (str(amrId).zfill(3),
                                    "(blank)" if before == "" else before,
                                    "(blank)" if isClear else newVal)
    _logOverride("AMR_%s" % str(amrId).zfill(3),
                 "AMR ComponentID Unassign" if isClear else "AMR ComponentID Set",
                 details, reason, _user(session))
    return {"ok": True, "message": "Committed: %s" % details}

def refreshNextWorkOrder(stationName, reason, session):
    """
    Refire the existing getOrder/updateRunSchedules on the owning back-end.
    BE1 for Frame/Boom/Paint, BE2 otherwise.
    """
    reason = (reason or "").strip()
    if not reason:
        return {"ok": False, "message": "A reason is required."}

    # getOrder() calls updateRunSchedules(stationName) on its first line, so a single
    # getOrder fire is sufficient -- the previously separate updateRunSchedules call
    # was a redundant second round trip / double schedule rebuild and is removed.
    _beExec("RunSchedule.getNextWorkOrder.getOrder",
            {"stationName": stationName}, stationName=stationName)

    _logOverride(stationName, "Run Schedule Refresh",
                 "Refired getNextWorkOrder", reason, _user(session))
    return {"ok": True, "message": "Refresh fired for %s." % stationName}

def validateSerialForAssembly(serialNumber):
    """
    Reuse existing checks: MachineConfig existence first, then assembly-scheduled
    check (CheckScheduledStatusBySerial). Returns a dict the UI renders.
    """
    serial = (serialNumber or "").strip()
    result = {"serial": serial, "ok": False, "messages": [], "alreadyScheduled": False,
              "priorNonConform": False, "reason": ""}
    if not serial:
        result["messages"].append("Enter a serial.")
        result["reason"] = "Enter a serial number."
        return result

    # 1) MachineConfig first (via the existing shared helper)
    mc = JcCommonOperations.getMachineConfigInfo(serial, ["ItemName"])
    inConfig = (mc is not None and len(mc) > 0)
    if not inConfig:
        result["messages"].append("Not found in MachineConfig.")
        result["reason"] = "%s is not in the MachineConfig table." % serial
        return result
    result["messages"].append("Exists in MachineConfig.")

    # 2) already scheduled in assembly?  (BE query via executeFunction -> dynamicQuery)
    cnt = _beQuery("RunSchedule/Assembly/Select/CheckScheduledStatusBySerial",
                   {"serialNumber": serial})
    cnt = int(cnt)
    if cnt > 0:
        result["alreadyScheduled"] = True
        result["messages"].append("Already scheduled in Assembly.")
        result["reason"] = "%s is already scheduled in Assembly." % serial
        return result
    result["messages"].append("Not already scheduled in Assembly.")
    result["reason"] = "Validated - ready to schedule."
    result["ok"] = True
    return result


def scheduleSerialForAssembly(serialNumber, legacy, reason, session):
    """Fire the EXISTING createAssemblyOrder on BE2 (same pattern the screen uses)."""
    serial = (serialNumber or "").strip()
    reason = (reason or "").strip()
    if not reason:
        return {"ok": False, "message": "A reason is required."}
    # createAssemblyOrder is BE2-resident (reads build slots from [TAGIO1] RawScheduleData
    # but is invoked on BE2 per the established manual-assembly screen pattern).
    # _beExec with no stationName routes to BE2 (getBackEnd_2).
    scheduleResult = _beExec(
        "RunSchedule.Assembly.Assembly.createAssemblyOrder",
        {"serialNumber": serial, "legacy": bool(legacy)})
    # Honor the BE return (True/False) instead of always reporting success.
    if scheduleResult is False:
        return {"ok": False, "message": "Backend declined to schedule %s for Assembly." % serial}
    _logOverride(serial, "Manual Assembly Schedule",
                 "createAssemblyOrder (legacy=%s)" % bool(legacy), reason, _user(session))
    return {"ok": True, "message": "Scheduled %s for Assembly." % serial}

def lookupComponent(idText):
    """
    Classify + locate a ComponentID/serial using EXISTING functions only.
      - starts with 'M' => NWG; else => serial (WG)
      - MachineConfig first (FE JcCommonOperations.getMachineConfigInfo)
      - serial -> NWGs (FE-direct JcCommonOperations.getNwgAssociatedItems -> {ItemName: ComponentID})
      - NWG -> serial (BE StationControl.findSerialNumber.findSerialNumber -> FinalSerialNumber str)
    Returns a dict for the UI.
    """
    val = (idText or "").strip()
    out = {"input": val, "isNWG": val.upper().startswith("M"), "found": False,
           "itemName": None, "itemNumber": None, "associations": None, "serial": None}
    if not val:
        return out

    mc = JcCommonOperations.getMachineConfigInfo(val, ["ItemName", "FinishedItemNumber"])
    if mc is None or len(mc) == 0:
        out["messages"] = "Not found in MachineConfig."
        return out
    out["found"] = True
    for row in mc:
        desc = row["Description"]
        if desc == "ItemName":
            out["itemName"] = row["Value"]
        elif desc == "FinishedItemNumber":
            out["itemNumber"] = row["Value"]

    if out["isNWG"]:
        # NWG -> which serial it is associated to (if any). The BE wrapper
        # findSerialNumber returns the FinalSerialNumber string directly (vs the raw
        # dataset the old FindWGSerialNumber query returned). FindWGSerialNumber is
        # BE-only, so this stays a BE call.
        out["serial"] = _beExec("StationControl.findSerialNumber.findSerialNumber",
                                {"NWGComponentID": val})
    else:
        # serial -> its NWGs via the EXISTING getNwgAssociatedItems wrapper, which
        # returns a ready {ItemName: ComponentID} dict (replaces the old raw-dataset
        # FindNwgsAssociatedWithSerialNumber query). Routed to the BE copy: the FE
        # JcCommonOperations.getNwgAssociatedItems copy calls its underlying query with
        # project='FactoryControl' (which has no RunSchedule/Select tree) and silently
        # returns {}; only the BE copy (project='FactoryGatewayEvents') actually resolves.
        assoc = _beExec("JcCommonOperations.getNwgAssociatedItems",
                        {"serialNumber": val})
        # getNwgAssociatedItems returns {ItemName: ComponentID}; the AssocTable binds to
        # a list of rows, so flatten to [{ItemName, ComponentID}, ...].
        out["associations"] = [{"ItemName": k, "ComponentID": v}
                               for k, v in (assoc or {}).items()]
    return out


def _fabStatusAtStation(componentID, stnType, stationLike):
    """
    Reuse existing selectStatus (TOP 1 status by component+station) via the BE.
    Returns an int status or None. selectStatus has fallbackValue '' (empty string),
    so coerce a missing/blank result to None -- otherwise the caller's integer guards
    (status == 1 / status == 2) would silently never match an empty string.
    """
    raw = _beQuery("RunSchedule/Fabrication/Select/selectStatus",
                   {"componentID": componentID, "stn": stationLike, "stnType": stnType},
                   stationName=stationLike)
    if raw is None or raw == "":
        return None
    return int(raw)


def scheduleNwgPrePaint(componentID, itemName, reason, session):
    """
    Reuse existing prepaint return queries. Block a duplicate Status-1 entry
    (per your direction) using the existing selectStatus read.
    Boom parts (Base/Fly/InnerMid/OuterMid) -> updateReturnBoomToPrePaint
    Frame -> updateReturnFrameToPrePaint
    """
    reason = (reason or "").strip()
    if not reason:
        return {"ok": False, "message": "A reason is required."}

    isFrame = (itemName == "Frame")
    stnType = "Frame" if isFrame else "Boom"
    # duplicate guard: already scheduled (status 1) at the prepaint station?
    stationLike = "TW07%" if isFrame else "TT17%"
    existing = _fabStatusAtStation(componentID, stnType, stationLike)
    if existing == 1:
        return {"ok": False, "message": "Already scheduled at prepaint (status 1); not adding a duplicate."}

    # Reuse the EXISTING PaintLoad return-to-prepaint functions.
    if isFrame:
        RunSchedule.PaintLoad.frameReturnToPrePaint(componentID)
    else:
        RunSchedule.PaintLoad.boomReturnToPrePaint(componentID)
    _logOverride(componentID, "NWG Schedule PrePaint",
                 "%s -> prepaint (%s)" % (componentID, stnType), reason, _user(session))
    return {"ok": True, "message": "%s scheduled to prepaint." % componentID}


def scheduleNwgPreAssembly(componentID, reason, session):
    """
    Reuse the EXISTING insertFabricationToPreAssemblyInventory (routes part-type
    internally, always TF000X, status 1). No carrier-group / upstream validation
    (per design). Fires on the owning back-end.
    """
    reason = (reason or "").strip()
    if not reason:
        return {"ok": False, "message": "A reason is required."}
    # This insert query lives in the FactoryGatewayEvents project. It routes part-type
    # internally and always targets TF000X with the passed status. (BE via dynamicQuery.)
    _beQuery("RunSchedule/PaintUnload/insert/insertFabricationToPreAssemblyInventory",
             {"componentID": componentID, "status": 1})
    _logOverride(componentID, "NWG Schedule PreAssembly",
                 "%s -> TF000X (status 1)" % componentID, reason, _user(session))
    return {"ok": True, "message": "%s scheduled to pre-assembly (TF000X)." % componentID}


def markNwgComplete(componentID, itemNumber, station, reason, session, warnIfWip=True):
    """
    Reuse existing updateCompleteTimestamp. Per your direction, WARN if the row is
    currently WIP (status 2) before completing - the UI passes the user's confirm
    by setting warnIfWip False on the second call.
    Returns {'ok','message','needsConfirm'}.
    """
    reason = (reason or "").strip()
    if not reason:
        return {"ok": False, "message": "A reason is required.", "needsConfirm": False}

    # Reconciled stnType derivation to MATCH the BE runQueryHelper convention
    # (startswith('TT') -> Boom, else Frame) so TF000X-staged parts pick the same
    # Fabrication_{stnType}WIP table the BE writes to.
    stnType = "Boom" if station.startswith("TT") else "Frame"

    if warnIfWip:
        status = _fabStatusAtStation(componentID, stnType, station)
        if status == 2:
            return {"ok": False, "needsConfirm": True,
                    "message": "%s is currently WIP (status 2) at %s. Confirm to mark complete." % (componentID, station)}

    # Reuse the EXISTING BE wrapper instead of the raw query: runQueryHelper sets the
    # required stn5 (=station[:5]) param the updateCompleteTimestamp query declares and
    # derives stnType the same way, so the named query binds cleanly. It is a thin alias
    # over updateCompleteTimestamp with NO side effects (unlike workComplete, which
    # cascades next-station scheduling -- deliberately NOT used here).
    _beExec("RunSchedule.Fabrication.UpdateWorkStatus.runQueryHelper",
            {"queryName": "UpdateWIPCompletedTimestamp",
             "componentID": componentID, "itemNumber": itemNumber,
             "station": station, "status": 3},
            stationName=station)
    _logOverride(station, "NWG Mark Complete",
                 "%s complete at %s" % (componentID, station), reason, _user(session))
    return {"ok": True, "message": "%s marked complete at %s." % (componentID, station), "needsConfirm": False}

def getScheduleEntries(idText):
    """
    Return EVERY WIP/schedule row (one per station) for a componentID or serial.
    Auto-routed: 'M'-prefix => NWG componentID -> Fabrication Frame+Boom WIP;
    otherwise => WG serial -> Assembly WIP. Both NEW FactoryControl queries are
    FE-direct (the WIP tables are on SCADA). Columns are unified:
    Station, Status (raw int), Item, ScheduledTimestamp, WIPTimestamp, CompletedTimestamp.
    Returns a list[dict] for the table binding.
    """
    val = (idText or "").strip()
    if not val:
        return []
    if val.upper().startswith("M"):
        ds = system.db.runNamedQuery(
            project=_PROJECT,
            path="station-control-center/getFabricationScheduleByComponent",
            parameters={"componentID": val})
    else:
        ds = system.db.runNamedQuery(
            project=_PROJECT,
            path="station-control-center/getAssemblyScheduleBySerial",
            parameters={"serialNumber": val})
    rows = []
    pds = system.dataset.toPyDataSet(ds)
    cols = list(pds.getColumnNames())
    for r in pds:
        d = {}
        for c in cols:
            d[c] = r[c]
        rows.append(d)
    return rows


def markSerialComplete(serialNumber, station, reason, session):
    """
    Mark a WG serial complete at an assembly station via the EXISTING BE query
    RunSchedule/Assembly/Update/WorkOrderComplete (sets CompletedTimestamp + Status=3
    on the latest row; leaves Non-Conformance 4/5 untouched). Reconciliation-style:
    no line cascade (unlike RunSchedule.Assembly.Assembly.workComplete).
    """
    reason = (reason or "").strip()
    if not reason:
        return {"ok": False, "message": "A reason is required.", "needsConfirm": False}
    _beQuery("RunSchedule/Assembly/Update/WorkOrderComplete",
             {"StationName": station, "SerialNumber": serialNumber, "Status": 3},
             stationName=station)
    _logOverride(station, "Serial Mark Complete",
                 "%s complete at %s" % (serialNumber, station), reason, _user(session))
    return {"ok": True, "message": "%s marked complete at %s." % (serialNumber, station), "needsConfirm": False}


def markScheduleComplete(idText, item, station, reason, session):
    """
    Router for the search page's 'Mark Complete at Station'. NWG componentID ('M')
    -> markNwgComplete (fab); WG serial -> markSerialComplete (assembly). warnIfWip is
    off because the user selects the entry from the table (status is already visible).
    """
    if (idText or "").strip().upper().startswith("M"):
        return markNwgComplete(idText, item, station, reason, session, warnIfWip=False)
    return markSerialComplete(idText, station, reason, session)
