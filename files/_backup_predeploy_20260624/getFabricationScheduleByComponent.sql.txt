-- NEW FactoryControl named query
-- Path:    station-control-center/getFabricationScheduleByComponent
-- Type:    Query (dataset)   DB: SCADA   FE-direct (WIP tables live on SCADA)
-- Param:   componentID  (String)
-- Returns: every fabrication WIP row (Frame + Boom) for an NWG componentID,
--          one row per station, with the raw Status (1=Scheduled,2=WIP,3=Complete,
--          4=Abort,5=Non-Conformance,7=Associated).
SELECT Station, Status, ItemNumber AS Item, ScheduledTimestamp, WIPTimestamp, CompletedTimestamp
FROM Fabrication_FrameWIP
WHERE ComponentID = :componentID
UNION ALL
SELECT Station, Status, ItemNumber AS Item, ScheduledTimestamp, WIPTimestamp, CompletedTimestamp
FROM Fabrication_BoomWIP
WHERE ComponentID = :componentID
ORDER BY ScheduledTimestamp
